pipeline {
    agent any
    
    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/ksalunkhegit/devops-exam.git'
            }
        }

        stage('Terraform Plan') {
            steps {
                script {
                    sh 'terraform init'
                    sh 'terraform plan'
                }
            }
        }

        stage('Apply Terraform') {
            steps {
                script {
                    sh 'terraform apply -auto-approve'
                }
            }
        }

        stage('Invoke Lambda') {
            steps {
                script {
                    sh 'aws lambda invoke --function-name myLambdaFunction output.txt'
                }
            }
        }
    }
}
